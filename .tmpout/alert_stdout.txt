"""ws3_alert.dtn — delay-tolerant Spray-and-Wait store-carry-forward queue (M8, offline path, C1).

When the cloud/SACHET link is down, alerts and readings are queued and rebroadcast over the LoRa mesh with
a bounded number of copies (spray count L), then delivered when a link returns. Pure stdlib.
"""
from __future__ import annotations

from dataclasses import dataclass, field


@dataclass
class DTNMessage:
    msg_id: str
    payload: bytes
    copies: int                    # remaining spray copies (binary spray-and-wait halves on relay)
    delivered: bool = False


@dataclass
class DTNQueue:
    L: int = 8                                   # initial copy budget
    store: dict[str, DTNMessage] = field(default_factory=dict)

    def enqueue(self, msg_id: str, payload: bytes) -> None:
        if msg_id not in self.store:
            self.store[msg_id] = DTNMessage(msg_id=msg_id, payload=payload, copies=self.L)

    def relay(self, msg_id: str):
        """Binary spray: hand off half the copies to a peer; keep the rest. Returns copies handed off."""
        m = self.store.get(msg_id)
        if not m or m.copies <= 1:
            return 0
        give = m.copies // 2
        m.copies -= give
        return give

    def deliver(self, msg_id: str) -> bool:
        """Mark delivered when a link to the destination is available."""
        m = self.store.get(msg_id)
        if not m:
            return False
        m.delivered = True
        return True

    def pending(self) -> list[str]:
        return [k for k, m in self.store.items() if not m.delivered]

"""Lightweight test runner (pytest is unavailable). Runs stdlib unittest over tests/.

Usage:  python run_tests.py            # all tests
        python run_tests.py pattern    # filter by filename glob (e.g. test_geo*)
"""
from __future__ import annotations

import sys
import unittest
from pathlib import Path

ROOT = Path(__file__).resolve().parent
SRC = ROOT / "src"
sys.path.insert(0, str(SRC))          # make `trinetra` importable
sys.path.insert(0, str(ROOT))         # make `tests` importable


def main() -> int:
    pattern = sys.argv[1] if len(sys.argv) > 1 else "test_*.py"
    loader = unittest.TestLoader()
    suite = loader.discover(str(ROOT / "tests"), pattern=pattern, top_level_dir=str(ROOT))
    result = unittest.TextTestRunner(verbosity=2).run(suite)
    return 0 if result.wasSuccessful() else 1


if __name__ == "__main__":
    raise SystemExit(main())

import unittest

from trinetra.common.config import Config
from trinetra.common.types import ExpertOutput, PhysicsClass, Hazard, Level

from trinetra.ws5_fusion_rl import decide, ShieldSignals
from trinetra.ws3_alert import build_cap, cap_fields_for_level, present, approve, DTNQueue


def _orange_decision():
    experts = [ExpertOutput(p=0.01, source="geo_phys", u=0.1, cls=PhysicsClass.A)]
    return decide(experts, ShieldSignals(FS=0.7), Config(), hazard=Hazard.LANDSLIDE, zone_id="UK-CHX-07")


class TestHumanGate(unittest.TestCase):
    def test_cap_refused_without_approval(self):
        d = _orange_decision()
        with self.assertRaises(SafetyViolation):
            build_cap(d, approval_token=None, sender="SDMA-UK")

    def test_cap_refused_with_bogus_token(self):
        d = _orange_decision()
        with self.assertRaises(SafetyViolation):
            build_cap(d, approval_token="totally-not-approved", sender="SDMA-UK")

    def test_cap_built_after_approval(self):
        d = _orange_decision()
        token = approve(d, operator_id="SDMA-UK-14")
        xml = build_cap(d, approval_token=token, sender="SDMA-UK", key=b"k")
        self.assertIn("<severity>Severe</severity>", xml)
        self.assertIn("UK-CHX-07", xml)
        self.assertIn("advisory", xml)
        self.assertIn("signature hmac-sha256:", xml)

    def test_green_has_no_public_cap(self):
        self.assertIsNone(cap_fields_for_level(Level.GREEN))
